**Goal:** Munshi speaks Hindi with an offline voice bundled in the APK, picked by the team from a hidden "Awaaz jaanch" screen that times and plays every candidate voice on our real lines.

**Architecture:**
- sherpa-onnx (pinned AAR, fetched and checksum-checked by Gradle) turns Devanagari text into PCM.
- A new `SherpaSpeaker` implements the existing `Speaker` interface: the same FIFO queue, a per-line PCM
  cache, and `AndroidSpeaker` as the fallback for anything it can't say.
- Candidate voices are discovered from folders. The test pushes them by `adb`; the release bundles one
  under `assets/voice/`, copied to app storage on first launch.
- `VoiceRig` (app-level) owns the speakers and the user's choice.

**Tech Stack:** Kotlin, Jetpack Compose, sherpa-onnx 1.13.8 (static onnxruntime AAR), Android `AudioTrack` (float PCM), DataStore, JUnit 4, kotlinx-coroutines-test, Paparazzi.

**Spec:** `docs/superpowers/specs/2026-10-25-on-device-ai-design.md` §6 (and §3 for the checked facts).

```kotlin
// sherpa-onnx and its voices are GitHub release files, not Maven artifacts: fetch each pinned file once
// into android/.cache and refuse it unless the SHA-256 matches.
fun sha256Of(f: File): String {
    val md = MessageDigest.getInstance("SHA-256")
    f.inputStream().use { s ->
        val buf = ByteArray(1 shl 16)
        while (true) { val n = s.read(buf); if (n < 0) break; md.update(buf, 0, n) }
    }
    return md.digest().joinToString("") { "%02x".format(it) }
}

fun fetchPinned(url: String, sha256: String, dest: File): File {
    if (dest.isFile && sha256Of(dest) == sha256) return dest
    dest.parentFile.mkdirs()
    val part = File(dest.path + ".part")
    logger.lifecycle("Downloading $url")
    URI(url).toURL().openStream().use { input -> part.outputStream().use { input.copyTo(it) } }
    val got = sha256Of(part)
    check(got == sha256) { "SHA-256 mismatch for $url: got $got, want $sha256" }
    check(part.renameTo(dest) || (dest.delete() && part.renameTo(dest))) { "Could not move $part to $dest" }
    return dest
}
```

```kotlin
package com.opuscube.munshi.voice

import com.k2fsa.sherpa.onnx.OfflineTts

/** Mono float PCM, the way sherpa-onnx returns it. */
class Pcm(val samples: FloatArray, val sampleRate: Int) {
    val durationMs: Long get() = if (sampleRate == 0) 0 else samples.size * 1000L / sampleRate
}

/** Text to audio. Blocking and slow: call it off the main thread. */
interface Synth {
    val id: String
    val label: String
    fun synth(text: String): Pcm
}

// The native engine loads on first use, so building one is cheap and a broken voice fails in synth().
class SherpaSynth(private val spec: VoiceSpec) : Synth {
    override val id: String get() = spec.id
    override val label: String get() = spec.label
    private val tts by lazy { OfflineTts(config = Voices.config(spec)) }
    override fun synth(text: String): Pcm = tts.generate(text, spec.sid, 1.0f).let { Pcm(it.samples, it.sampleRate) }
}
```

```kotlin
package com.opuscube.munshi.voice

import com.k2fsa.sherpa.onnx.OfflineTtsConfig
import com.k2fsa.sherpa.onnx.OfflineTtsKokoroModelConfig
import com.k2fsa.sherpa.onnx.OfflineTtsModelConfig
import com.k2fsa.sherpa.onnx.OfflineTtsVitsModelConfig
import java.io.File

/** One voice on disk, as unpacked from a sherpa-onnx `tts-models` archive. */
sealed interface VoiceSpec {
    val id: String
    val label: String
    val dir: File
    val sid: Int

    data class Piper(override val dir: File) : VoiceSpec {
        override val id: String get() = dir.name
        override val label: String get() = "Piper · " + dir.name.removePrefix("vits-piper-hi_IN-").substringBefore('-')
        override val sid: Int get() = 0
    }

    data class Kokoro(override val dir: File, override val sid: Int, val voice: String) : VoiceSpec {
        override val id: String get() = "${dir.name}#$sid"
        override val label: String get() = "Kokoro · $voice"
    }
}

object Voices {
    /** Kokoro v1.0 speaker ids of its Hindi voices (sherpa-onnx Kokoro docs). */
    val KOKORO_HINDI = listOf(31 to "hf_alpha", 33 to "hm_omega")

    fun discover(root: File?): List<VoiceSpec> =
        root?.listFiles().orEmpty().filter { it.isDirectory }.sortedBy { it.name }.flatMap { d ->
            val complete = d.onnx() != null && File(d, "tokens.txt").isFile && File(d, "espeak-ng-data").isDirectory
            when {
                !complete -> emptyList()
                d.name.startsWith("vits-piper-hi_IN-") -> listOf(VoiceSpec.Piper(d))
                d.name.startsWith("kokoro") && File(d, "voices.bin").isFile -> KOKORO_HINDI.map { (sid, v) -> VoiceSpec.Kokoro(d, sid, v) }
                else -> emptyList()
            }
        }

    // Files on disk, no AssetManager: espeak-ng needs a real directory for its data.
    fun config(spec: VoiceSpec): OfflineTtsConfig {
        val model = spec.dir.onnx()!!.path
        val tokens = File(spec.dir, "tokens.txt").path
        val data = File(spec.dir, "espeak-ng-data").path
        val m = when (spec) {
            is VoiceSpec.Piper -> OfflineTtsModelConfig(
                vits = OfflineTtsVitsModelConfig(model = model, tokens = tokens, dataDir = data),
                numThreads = 2, provider = "cpu",
            )
            is VoiceSpec.Kokoro -> OfflineTtsModelConfig(
                kokoro = OfflineTtsKokoroModelConfig(model = model, voices = File(spec.dir, "voices.bin").path, tokens = tokens, dataDir = data, lang = "hi"),
                numThreads = 2, provider = "cpu",
            )
        }
        return OfflineTtsConfig(model = m)
    }

    private fun File.onnx(): File? = listFiles()?.firstOrNull { it.isFile && it.name.endsWith(".onnx") }
}
```

```kotlin
package com.opuscube.munshi.voice

import android.media.AudioAttributes
import android.media.AudioFormat
import android.media.AudioTrack
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.delay
import kotlinx.coroutines.withContext
import kotlinx.coroutines.withTimeoutOrNull

interface Player {
    /** Plays to the end, or until stop(). */
    suspend fun play(pcm: Pcm)
    fun stop()
}

class AudioTrackPlayer : Player {
    @Volatile private var track: AudioTrack? = null

    override suspend fun play(pcm: Pcm) = withContext(Dispatchers.IO) {
        val min = AudioTrack.getMinBufferSize(pcm.sampleRate, AudioFormat.CHANNEL_OUT_MONO, AudioFormat.ENCODING_PCM_FLOAT)
        val t = AudioTrack.Builder()
            .setAudioAttributes(AudioAttributes.Builder().setUsage(AudioAttributes.USAGE_ASSISTANT).setContentType(AudioAttributes.CONTENT_TYPE_SPEECH).build())
            .setAudioFormat(AudioFormat.Builder().setEncoding(AudioFormat.ENCODING_PCM_FLOAT).setSampleRate(pcm.sampleRate).setChannelMask(AudioFormat.CHANNEL_OUT_MONO).build())
            .setTransferMode(AudioTrack.MODE_STREAM)
            .setBufferSizeInBytes(maxOf(min, 8192))
            .build()
        track = t
        try {
            t.play()
            var off = 0
            while (off < pcm.samples.size && track === t) {
                val n = t.write(pcm.samples, off, minOf(4096, pcm.samples.size - off), AudioTrack.WRITE_BLOCKING)
                if (n <= 0) break
                off += n
            }
            // write() returns once the data is buffered; wait for the speaker to actually reach the end.
            withTimeoutOrNull(pcm.durationMs + 1000) {
                while (track === t && t.playbackHeadPosition < pcm.samples.size) delay(20)
            }
        } finally {
            runCatching { t.stop() }
            t.release()
            if (track === t) track = null
        }
    }

    override fun stop() {
        val t = track ?: return
        track = null
        runCatching { t.pause(); t.flush() }
    }
}
```

```kotlin
package com.opuscube.munshi.voice

import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.CoroutineDispatcher
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.sync.Mutex
import kotlinx.coroutines.sync.withLock
import kotlinx.coroutines.withContext

// The bundled Hindi voice. One FIFO queue like AndroidSpeaker; it speaks the Devanagari line and hands
// anything it cannot say (no Devanagari, no voice, a native error, silence) to the fallback speaker.
class SherpaSpeaker(
    private val player: Player,
    private val fallback: Speaker,
    private val work: CoroutineDispatcher = Dispatchers.Default,
    private val cacheSize: Int = 40,
) : Speaker {
    @Volatile private var synth: Synth? = null
    private val queue = Mutex()
    private val synthLock = Mutex()   // one native engine, one sentence at a time
    private val cache = LinkedHashMap<String, Pcm>(16, 0.75f, true)

    val voiceId: String? get() = synth?.id

    /** Switches voice (null = fallback only) and drops audio made with the old one. */
    fun use(s: Synth?) { synth = s; synchronized(cache) { cache.clear() } }

    override suspend fun say(line: Line) = queue.withLock {
        val pcm = line.spoken?.let { pcmOrNull(it) }
        if (pcm == null) fallback.say(line) else player.play(pcm)
    }

    override suspend fun prewarm(lines: List<Line>) { for (l in lines) l.spoken?.let { pcmOrNull(it) } }

    override fun stop() { player.stop(); fallback.stop() }

    private suspend fun pcmOrNull(text: String): Pcm? {
        val s = synth ?: return null
        synchronized(cache) { cache[text] }?.let { return it }
        val pcm = try {
            synthLock.withLock { withContext(work) { s.synth(text) } }
        } catch (e: Throwable) {
            if (e is CancellationException) throw e
            return null   // includes UnsatisfiedLinkError from a device without the native library
        }
        if (pcm.samples.isEmpty()) return null
        synchronized(cache) {
            cache[text] = pcm
            if (cache.size > cacheSize) cache.remove(cache.keys.first())
        }
        return pcm
    }
}
```

```kotlin
    Line("₹50,000 aapke khaate mein aa gaye. Roz ₹400 settlement se katega.", "₹50,000 आपके खाते में आ गए। रोज़ ₹400 सेटलमेंट से कटेगा।"),

private const val MIN_AUDIO_MS = 300L

sealed interface LabStatus {
    data object Untested : LabStatus
    data object Testing : LabStatus
    data object Baseline : LabStatus
    data class Measured(val firstAudioMs: Long, val rtf: Double) : LabStatus
    data class NotSupported(val why: String) : LabStatus
}

data class LabRow(val id: String, val label: String, val status: LabStatus)

// "Awaaz jaanch": time every candidate voice on the real lines, play any line, so the team can pick one.
class VoiceLab(
    private val candidates: List<Synth>,
    private val player: Player,
    private val baseline: Speaker,
    private val lines: List<Line> = LAB_LINES,
    private val work: CoroutineDispatcher = Dispatchers.Default,
    private val nanos: () -> Long = System::nanoTime,
) {
    private val audio = mutableMapOf<Pair<String, Int>, Pcm>()
    private val _rows = MutableStateFlow(
        listOf(LabRow(VoiceRig.ANDROID_TTS, "Android TTS (hi-IN)", LabStatus.Baseline)) +
            candidates.map { LabRow(it.id, it.label, LabStatus.Untested) },
    )
    val rows: StateFlow<List<LabRow>> = _rows

    suspend fun measureAll() {
        for (c in candidates) {
            set(c.id, LabStatus.Testing)
            set(c.id, measure(c))
        }
    }

    suspend fun play(id: String, line: Int) {
        val l = lines[line]
        if (id == VoiceRig.ANDROID_TTS) { baseline.say(l); return }
        val c = candidates.firstOrNull { it.id == id } ?: return
        val pcm = audio[id to line] ?: synthOrNull(c, l.spoken!!) ?: return
        player.play(pcm)
    }

    fun stop() { player.stop(); baseline.stop() }

    // First audio = time to synthesise line 1 (we synthesise whole lines). RTF = synth time / audio time.
    private suspend fun measure(c: Synth): LabStatus {
        var synthMs = 0L; var audioMs = 0L; var firstMs = 0L
        lines.forEachIndexed { i, line ->
            val t0 = nanos()
            val pcm = try { withContext(work) { c.synth(line.spoken!!) } } catch (e: Throwable) {
                if (e is CancellationException) throw e
                return LabStatus.NotSupported(e.message ?: e.javaClass.simpleName)
            }
            val ms = (nanos() - t0) / 1_000_000
            if (pcm.durationMs < MIN_AUDIO_MS) return LabStatus.NotSupported("no speech for line ${i + 1}")
            if (i == 0) firstMs = ms
            synthMs += ms; audioMs += pcm.durationMs
            audio[c.id to i] = pcm
        }
        return LabStatus.Measured(firstMs, synthMs.toDouble() / audioMs)
    }

    private suspend fun synthOrNull(c: Synth, text: String): Pcm? = try { withContext(work) { c.synth(text) } } catch (e: Throwable) {
        if (e is CancellationException) throw e
        null
    }

    private fun set(id: String, s: LabStatus) = _rows.update { rows -> rows.map { if (it.id == id) it.copy(status = s) else it } }
}
```

```powershell
$base = 'https://github.com/k2-fsa/sherpa-onnx/releases/download/tts-models'
$candidates = [ordered]@{
  'vits-piper-hi_IN-pratham-medium-fp16'    = '76b8f4404825e1c6a588bc3cee4b2e8eef1cfe1f785950f4a1b0cdf8d1050c97'

  'vits-piper-hi_IN-rohan-medium-fp16'      = 'b313549eca6c20b52c4a58d2773b3e79f27d347d2f020338b903ff1bb8e45b44'

}
$work = Join-Path $env:TEMP 'munshi-voices'
New-Item -ItemType Directory -Force $work | Out-Null
$target = '/sdcard/Android/data/com.opuscube.munshi/files/voices'
& $adb shell mkdir -p $target
foreach ($name in $candidates.Keys) {
  $archive = Join-Path $work "$name.tar.bz2"
  $want = $candidates[$name]
  if (-not (Test-Path $archive) -or (Get-FileHash $archive -Algorithm SHA256).Hash.ToLower() -ne $want) {
```

```kotlin
package com.opuscube.munshi.voice

import android.content.res.AssetManager
import java.io.File
import java.io.InputStream

interface AssetSource {
    /** Children of a folder; empty for a file or a missing path. */
    fun list(path: String): List<String>
    fun open(path: String): InputStream
}

class AndroidAssets(private val am: AssetManager) : AssetSource {
    override fun list(path: String) = am.list(path)?.toList().orEmpty()
    override fun open(path: String): InputStream = am.open(path)
}

object VoiceAssets {
    /** Copies assets/voice/** into [dest] once per [version]: espeak-ng and onnxruntime need real files. */
    fun install(src: AssetSource, dest: File, version: String) {
        val marker = File(dest, ".installed")
        if (marker.isFile && marker.readText() == version) return
        dest.deleteRecursively()
        dest.mkdirs()
        for (child in src.list("voice")) copy(src, "voice/$child", File(dest, child))
        marker.writeText(version)
    }

    private fun copy(src: AssetSource, path: String, to: File) {
        val children = src.list(path)
        if (children.isEmpty()) {
            to.parentFile?.mkdirs()
            src.open(path).use { input -> to.outputStream().use { input.copyTo(it) } }
            return
        }
        to.mkdirs()
        for (c in children) copy(src, "$path/$c", File(to, c))
    }
}
```
